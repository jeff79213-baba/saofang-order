> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 打造手機優先的兩棟民宿掃房順序網頁 App，格子呈現房間狀態，依權重自動算出每層掃房順序，櫃台／房務即時同步。

**Architecture:** React 18 + TS + Vite 單頁應用。純函式（`ordering.ts` / `time.ts` / `rooms.ts` / `dayKey.ts`）承載所有計算邏輯並用 Vitest 覆蓋；資料存取抽象在 `RoomRepo` 介面，提供 `MockRepo`（localStorage，DEMO）與 `FirebaseRepo`（Firestore onSnapshot）兩種實作，依環境變數選擇。

**Tech Stack:** React 18、TypeScript、Vite、Tailwind CSS、Vitest、Playwright、Firebase Auth/Firestore/Hosting。

## Global Constraints

- 框架固定 React 18 + TypeScript + Vite；樣式用 Tailwind CSS。
- **禁止**引入任何 UI 元件庫、狀態管理庫（Redux/Zustand）、日期庫（date-fns/dayjs）、icon 套件。
- Firestore 頂層集合一律加 `sq-` 前綴：`sq-staff`、`sq-settings`、`sq-rooms`、`sq-days`。
- localStorage key 一律加 `sq-` 前綴。
- 樓層固定 `[1,2,3,5]`，**全系統不得出現 4 樓**。
- 預設房數：一館 A 每層 2 間（共 8）、二館 B 每層 12 間（共 48），合計 56，皆可於設定頁調整（0–20）。
- **v1 沒有「熱門房」功能**（無 `hot` 欄位、無 ⭐）。
- Hosting site 固定 `saofang-sk`；`firebase.json` 必須含 `"site": "saofang-sk"`。
- 角色兩種：`front`（櫃台）、`housekeeping`（房務）。房務唯讀＋只能打掃。
- 手機優先：390×844 直式二館單層 12 間不可橫向捲動。
- 交件標準：`npm run dev` 可操作（無 Firebase 設定走 DEMO）、`npm run test` 全綠。

import type { BuildingId, RoomMeta, Settings } from '../types'
import { BUILDINGS } from '../types'

export function roomNo(floor: number, seq: number): string {
  return `${floor * 100 + seq}`
}

export function buildRoomList(
  layout: Settings['layout'],
  floors: number[],
): RoomMeta[] {
  const out: RoomMeta[] = []
  for (const building of BUILDINGS) {
    for (const floor of floors) {
      const count = layout[building]?.[String(floor)] ?? 0
      for (let seq = 1; seq <= count; seq++) {
        const no = roomNo(floor, seq)
        out.push({ id: `${building}-${no}`, building, floor, no, enabled: true })
      }
    }
  }
  return out
}

export function buildingRoomCount(layout: Settings['layout'], building: BuildingId): number {
  const floorsMap = layout[building] ?? {}
  return Object.values(floorsMap).reduce((sum, n) => sum + (n ?? 0), 0)
}
```

export function todayKey(now: Date = new Date()): string {
  const parts = new Intl.DateTimeFormat('en-CA', {
    timeZone: 'Asia/Taipei',
    year: 'numeric',
    month: '2-digit',
    day: '2-digit',
  }).formatToParts(now)
  const get = (t: string) => parts.find(p => p.type === t)!.value
  return `${get('year')}-${get('month')}-${get('day')}`
}

export function nextDayKey(dayKey: string): string {
  const [y, m, d] = dayKey.split('-').map(Number)
  const dt = new Date(Date.UTC(y, m - 1, d))
  dt.setUTCDate(dt.getUTCDate() + 1)
  return `${dt.getUTCFullYear()}-${String(dt.getUTCMonth() + 1).padStart(2, '0')}-${String(dt.getUTCDate()).padStart(2, '0')}`
}

export function prevDayKey(dayKey: string): string {
  const [y, m, d] = dayKey.split('-').map(Number)
  const dt = new Date(Date.UTC(y, m - 1, d))
  dt.setUTCDate(dt.getUTCDate() - 1)
  return `${dt.getUTCFullYear()}-${String(dt.getUTCMonth() + 1).padStart(2, '0')}-${String(dt.getUTCDate()).padStart(2, '0')}`
}

import type { RoomState } from '../types'

export function tier(r: RoomState): number | null {
  if (r.status === 'occupied' || r.status === 'clean') return null
  if (r.status === 'arriving' && r.checkInTime) return 1
  if (r.status === 'arriving') return 2
  if (r.status === 'dirty' && r.checkedOutAt) return 3
  if (r.status === 'dirty') return 4
  return null
}

const cmpNo = (a: RoomState, b: RoomState) =>
  a.no.localeCompare(b.no, 'en', { numeric: true })

export function buildQueue(rooms: RoomState[]): Array<{ room: RoomState; seq: number }> {
  return rooms
    .filter(r => tier(r) !== null)
    .sort((a, b) => {
      const ta = tier(a)!, tb = tier(b)!
      if (ta !== tb) return ta - tb
      if (ta === 1) {
        if (a.checkInTime !== b.checkInTime) return a.checkInTime! < b.checkInTime! ? -1 : 1
        return cmpNo(a, b)
      }
      return cmpNo(a, b)
    })
    .map((room, i) => ({ room, seq: i + 1 }))
}
```

export function parseDigits(raw: string): { time: string | null; error?: string } {
  const d = raw.replace(/\D/g, '').slice(0, 4)
  if (d.length < 3) return { time: null, error: '請輸入 4 位數字' }
  const s = d.padStart(4, '0')
  let hh = Number(s.slice(0, 2))
  const mm = Number(s.slice(2, 4))
  if (mm > 59) return { time: null, error: '分鐘格式錯誤' }
  if (hh > 23) return { time: null, error: '小時格式錯誤' }
  if (hh >= 1 && hh <= 11) hh += 12
  return { time: `${String(hh).padStart(2, '0')}:${String(mm).padStart(2, '0')}` }
}

export function formatZh(time: string): string {
  const [hStr, mStr] = time.split(':')
  const h = Number(hStr)
  const ampm = h < 12 ? '上午' : '下午'
  let h12 = h % 12
  if (h12 === 0) h12 = 12
  return `${ampm} ${h12}:${mStr}`
}

export function shiftAmPm(time: string, dir: 1 | -1): string {
  const [hStr, mStr] = time.split(':')
  const h = (Number(hStr) + dir * 12 + 24) % 24
  return `${String(h).padStart(2, '0')}:${mStr}`
}

      if (!daySubs.has(dayKey)) daySubs.set(dayKey, new Set())

      writeDay(dayKey, list)

      await this.resetDay(prevDayKey, dayKey)

      cb(structuredClone(rooms))
      return () => { roomSubs.delete(cb) }

import { useRef } from 'react'
import type { Role, RoomState } from '../types'
import { NextGuestTag } from './NextGuestTag'

interface Props {
  room: RoomState
  seq: number | null
  role: Role
  mode: 'normal' | 'clean'
  pending: boolean
  onCardClick: (room: RoomState) => void
  onLongPressTime: (room: RoomState) => void
  onToggleNextGuest: (room: RoomState) => void
}

const LONG_MS = 1000

export function RoomCard({ room, seq, role, mode, pending, onCardClick, onLongPressTime, onToggleNextGuest }: Props) {
  const pressRef = useRef<{ timer: number; started: boolean; cancelled: boolean; startX: number; startY: number } | null>(null)
  const suppressRef = useRef(false)
  const canSetTime = role === 'front'

  const skin = (() => {
    if (room.status === 'occupied') return { cls: 'bg-amber-400 shadow-[0_0_16px_rgba(251,191,36,.55)] text-amber-950', border: '' }
    if (room.status === 'arriving') return { cls: 'bg-white text-red-600', border: 'border-2 border-red-500' }
    if (room.status === 'dirty' && room.checkedOutAt) return { cls: 'bg-white text-slate-900', border: 'border-2 border-blue-500' }
    if (room.status === 'dirty') return { cls: 'bg-slate-50 text-slate-700', border: 'border border-slate-300' }
    return { cls: 'bg-emerald-50 text-emerald-700 opacity-60', border: 'border border-emerald-300' }
  })()

  const center =
    room.status === 'occupied' ? '入住中'
    : room.status === 'clean' ? '✓ 已打掃'
    : seq !== null ? `待掃 ${seq}` : ''

  function startPress(e: React.PointerEvent) {
    if (!canSetTime) return
    const p = { timer: 0, started: true, cancelled: false, startX: e.clientX, startY: e.clientY }
    pressRef.current = p
    p.timer = window.setTimeout(() => {
      if (!p.cancelled) {
        suppressRef.current = true
        navigator.vibrate?.(30)
        onLongPressTime(room)
      }
    }, LONG_MS)
  }
  function movePress(e: React.PointerEvent) {
    const p = pressRef.current
    if (!p) return
    if (Math.abs(e.clientX - p.startX) > 10 || Math.abs(e.clientY - p.startY) > 10) endPress()
  }
  function endPress() {
    const p = pressRef.current
    if (p) { clearTimeout(p.timer); p.cancelled = true }
    pressRef.current = null
  }

  return (
    <div
      className={`relative flex min-h-[88px] cursor-pointer select-none flex-col justify-between rounded-2xl p-2 ${skin.cls} ${skin.border} ${pending ? 'animate-pulse opacity-70' : ''} ${mode === 'clean' && room.status !== 'occupied' ? 'active:scale-95' : ''}`}
      style={{ touchAction: 'manipulation' }}
      onClick={() => {
        if (suppressRef.current) { suppressRef.current = false; return }
        if (mode === 'clean' && room.status === 'occupied') return
        onCardClick(room)
      }}
    >
      <div className="flex items-start justify-between">
        <span className="text-xl font-bold tabular-nums">{room.no}</span>
      </div>
      <div className="text-center text-2xl font-black tabular-nums">{center}</div>
      <div className="flex flex-col items-center gap-1">
        <div
          className="w-full text-center text-[11px] tabular-nums text-slate-500"
          onPointerDown={startPress}
          onPointerUp={endPress}
          onPointerCancel={endPress}
          onPointerMove={movePress}
        >
          {room.checkInTime ?? '----'}
        </div>
        <NextGuestTag on={room.nextGuest} editable={role === 'front'} onToggle={() => onToggleNextGuest(room)} />
      </div>
    </div>
  )
}

import { useEffect, useRef, useState } from 'react'
import type { RoomState } from '../types'
import { formatZh, parseDigits, shiftAmPm } from '../lib/time'

export function TimeModal({ room, onConfirm, onClear, onClose }: {
  room: RoomState
  onConfirm: (time: string) => void
  onClear: () => void
  onClose: () => void
}) {
  const [raw, setRaw] = useState(room.checkInTime ? room.checkInTime.replace(':', '') : '')
  const [err, setErr] = useState<string | null>(null)
  const inputRef = useRef<HTMLInputElement>(null)

  useEffect(() => {
    inputRef.current?.focus()
    inputRef.current?.select()
  }, [])

  const parsed = parseDigits(raw)

  function confirm() {
    if (parsed.error || !parsed.time) { setErr(parsed.error ?? '時間錯誤'); return }
    onConfirm(parsed.time)
  }
  function shift(dir: 1 | -1) {
    const base = parsed.time ?? '12:00'
    setRaw(shiftAmPm(base, dir).replace(':', ''))
    setErr(null)
  }

  return (
    <div className="fixed inset-0 z-50 flex items-end bg-black/40" onClick={onClose}>
      <div className="w-full rounded-t-3xl bg-white p-5" onClick={e => e.stopPropagation()}>
        <h3 className="text-base font-bold text-slate-900">{room.no} 入住時間</h3>
        <input
          ref={inputRef}
          inputMode="numeric"
          maxLength={4}
          value={raw}
          onChange={e => { setRaw(e.target.value); setErr(null) }}
          className="mt-4 w-full rounded-xl border border-slate-300 p-3 text-center text-3xl font-black tabular-nums tracking-widest"
          placeholder="HHMM"
        />
        <div className="mt-2 text-center text-lg font-bold text-indigo-600 tabular-nums">
          {parsed.time ?? '--:--'}
        </div>
        <div className="mt-1 text-center text-xs text-slate-500">
          {parsed.time ? formatZh(parsed.time) : ''}
        </div>
        {err && <div className="mt-2 text-center text-sm text-red-600">{err}</div>}
        <div className="mt-4 flex justify-center gap-2">
          <button className="rounded-lg border border-slate-300 px-3 py-2 text-sm" onClick={() => shift(-1)}>上午</button>
          <button className="rounded-lg border border-slate-300 px-3 py-2 text-sm" onClick={() => shift(1)}>下午</button>
        </div>
        <div className="mt-5 flex justify-between">
          <button className="rounded-lg px-4 py-2 text-sm text-red-600" onClick={onClear}>清除</button>
          <div className="flex gap-2">
            <button className="rounded-lg px-4 py-2 text-sm text-slate-600" onClick={onClose}>取消</button>

          </div>
        </div>
      </div>
    </div>
  )

import { useEffect, useMemo, useRef, useState } from 'react'
import type { RoomMeta, RoomState, Settings } from '../types'
import { DEFAULT_FLOORS, DEFAULT_LAYOUT } from '../types'
import { getRepo } from '../lib/repo'
import { todayKey } from '../lib/dayKey'

export function useRooms(dayKey: string) {
  const repo = getRepo()
  const [dayRooms, setDayRooms] = useState<RoomState[]>([])
  const [roomMetas, setRoomMetas] = useState<RoomMeta[]>([])
  const [settings, setSettings] = useState<Settings>({ floors: DEFAULT_FLOORS, layout: DEFAULT_LAYOUT })
  const [loading, setLoading] = useState(true)
  const inited = useRef(false)

  useEffect(() => {
    const repo = getRepo()
    repo.getSettings().then(setSettings)
    const un1 = repo.subscribeRooms(setRoomMetas)
    const un2 = repo.subscribe(dayKey, rooms => { setDayRooms(rooms); setLoading(false) })
    if (!inited.current) {
      inited.current = true
      const prev = new Date(Date.now() - 86400000)
      repo.ensureDay(dayKey, todayKey(prev)).catch(() => {})
    }
    return () => { un1(); un2() }
  }, [dayKey])

  const enabledIds = useMemo(() => new Set(roomMetas.filter(r => r.enabled).map(r => r.id)), [roomMetas])
  const rooms = useMemo(
    () => dayRooms.filter(r => enabledIds.has(r.id)),
    [dayRooms, enabledIds],
  )
  return { rooms, dayRooms, roomMetas, settings, loading, repo }
}

```tsx
import { useMemo, useState } from 'react'
import type { BuildingId, RoomState } from '../types'
import { BUILDINGS } from '../types'
import { buildQueue } from '../lib/ordering'
import { nextDayKey, todayKey } from '../lib/dayKey'
import { HeaderBar } from '../components/HeaderBar'
import { FloorSection } from '../components/FloorSection'
import { RoomCard } from '../components/RoomCard'
import { TimeModal } from '../components/TimeModal'
import { ConfirmDialog } from '../components/ConfirmDialog'
import { ToastHost, type ToastMsg } from '../components/Toast'

interface Props {
  rooms: RoomState[]
  settings: { floors: number[] }
  role: 'front' | 'housekeeping'
  isDemo: boolean
  repo: ReturnType<typeof import('../lib/repo').getRepo>
  dayKey: string
  onOpenSettings: () => void
}

export function MainPage({ rooms, settings, role, isDemo, repo, dayKey, onOpenSettings }: Props) {
  const [building, setBuilding] = useState<BuildingId>('A')
  const [mode, setMode] = useState<'normal' | 'clean'>(role === 'housekeeping' ? 'clean' : 'normal')
  const [pending, setPending] = useState<Set<string>>(new Set())
  const [toasts, setToasts] = useState<ToastMsg[]>([])
  const [timeRoom, setTimeRoom] = useState<RoomState | null>(null)
  const [confirm, setConfirm] = useState<null | { title: string; message: string; onOk: () => void }>(null)

  const effectiveMode = role === 'housekeeping' ? 'clean' : mode

  function toast(text: string) {
    const id = Date.now() + Math.random()
    setToasts(t => [...t, { id, text }])
    setTimeout(() => setToasts(t => t.filter(x => x.id !== id)), 2000)
  }
  async function send(patches: Array<{ roomId: string; patch: Partial<RoomState> }>) {
    const ids = patches.map(p => p.roomId)
    setPending(prev => new Set([...prev, ...ids]))
    try {
      await repo.bulkUpdate(patches)
    } catch {
      toast('儲存失敗，請重試')
    } finally {
      setPending(prev => { const n = new Set(prev); ids.forEach(i => n.delete(i)); return n })
    }
  }

  const byFloor = useMemo(() => {
    const map = new Map<number, RoomState[]>()
    for (const r of rooms.filter(r => r.building === building)) {
      if (!map.has(r.floor)) map.set(r.floor, [])
      map.get(r.floor)!.push(r)
    }
    return map
  }, [rooms, building])

  function seqMap(floorRooms: RoomState[]) {
    const m = new Map<string, number>()
    for (const { room, seq } of buildQueue(floorRooms)) m.set(room.id, seq)
    return m
  }

  function onCardClick(room: RoomState) {
    if (effectiveMode === 'clean') {
      if (room.status === 'occupied') { toast('入住中不可掃'); return }
      if (room.status === 'clean') {
        send([{ roomId: room.id, patch: { status: room.prevStatus ?? 'dirty', prevStatus: null } }])
      } else {
        send([{ roomId: room.id, patch: { prevStatus: room.status, status: 'clean' } }])
      }
      return
    }
    if (role !== 'front') return
    if (room.status === 'occupied') {
      send([{ roomId: room.id, patch: {
        status: room.nextGuest ? 'arriving' : 'dirty',
        checkedOutAt: Date.now(), checkInTime: null, prevStatus: null,
      } }])
    } else {
      send([{ roomId: room.id, patch: {
        status: 'occupied', nextGuest: false, checkedOutAt: null, prevStatus: null,
      } }])
    }
  }

  function onToggleNextGuest(room: RoomState) {
    if (role !== 'front') return
    const next = !room.nextGuest
    const patch: Partial<RoomState> = { nextGuest: next }
    if (next && room.status !== 'occupied') patch.status = 'arriving'
    if (!next && room.status === 'arriving' && !room.checkInTime) patch.status = 'dirty'
    send([{ roomId: room.id, patch }])
  }

  function onConfirmTime(time: string) {
    if (!timeRoom) return
    const patch: Partial<RoomState> = { checkInTime: time }
    if (timeRoom.status === 'dirty' || timeRoom.status === 'clean') patch.status = 'arriving'
    send([{ roomId: timeRoom.id, patch }])
    setTimeRoom(null)
  }
  function onClearTime() {
    if (!timeRoom) return
    send([{ roomId: timeRoom.id, patch: { checkInTime: null } }])
    setTimeRoom(null)
  }

  function onFillHouse() {
    const inBuilding = rooms.filter(r => r.building === building)
    const skipped = inBuilding.filter(r => r.nextGuest).length
    setConfirm({
      title: `一鍵滿房 ${building === 'A' ? '一館' : '二館'}`,
      message: `${inBuilding.length} 間將全部亮燈，設有「後續入住」的房間會維持待入住。確定？`,
      onOk: () => {
        setConfirm(null)
        const patches = inBuilding.map(r => r.nextGuest
          ? { roomId: r.id, patch: { status: 'arriving' as const, checkedOutAt: null } }
          : { roomId: r.id, patch: { status: 'occupied' as const, checkedOutAt: null } })
        send(patches)
        toast(`${building === 'A' ? '一館' : '二館'}已滿房${skipped ? `（跳過 ${skipped} 間後續入住）` : ''}`)
      },
    })
  }

  function onResetDay() {
    const target = nextDayKey(dayKey)
    setConfirm({
      title: '換日',
      message: `建立 ${target} 的一天？今天的紀錄會保留。`,
      onOk: async () => {
        setConfirm(null)
        await repo.resetDay(dayKey, target)
        toast(`已建立 ${target}`)
        window.location.reload()
      },
    })
  }

  const columns = building === 'A'
    ? 'grid-cols-2 sm:grid-cols-4'
    : 'grid-cols-4 sm:grid-cols-6'

  return (
    <div className="min-h-full bg-slate-100 pb-10">
      <HeaderBar
        building={building} onBuilding={setBuilding} role={role}
        mode={effectiveMode} onMode={() => setMode(m => m === 'clean' ? 'normal' : 'clean')}
        dayKey={dayKey} totalRooms={rooms.length} isDemo={isDemo}
        onFillHouse={onFillHouse} onResetDay={onResetDay} onSettings={onOpenSettings}
      />
      <main className="px-4 py-4">
        {settings.floors.map(floor => {
          const floorRooms = byFloor.get(floor) ?? []
          if (floorRooms.length === 0) return null
          const sm = seqMap(floorRooms)
          return (
            <FloorSection
              key={floor} building={building} floor={floor} rooms={floorRooms} columns={columns}
              renderCard={room => (
                <RoomCard
                  key={room.id} room={room} seq={sm.get(room.id) ?? null}
                  role={role} mode={effectiveMode} pending={pending.has(room.id)}
                  onCardClick={onCardClick} onLongPressTime={setTimeRoom} onToggleNextGuest={onToggleNextGuest}
                />
              )}
            />
          )
        })}
      </main>
      {timeRoom && (
        <TimeModal room={timeRoom} onConfirm={onConfirmTime} onClear={onClearTime} onClose={() => setTimeRoom(null)} />
      )}
      <ConfirmDialog
        open={!!confirm} title={confirm?.title ?? ''} message={confirm?.message ?? ''}
        onConfirm={() => confirm?.onOk()} onCancel={() => setConfirm(null)}
      />
      <ToastHost toasts={toasts} />
    </div>
  )
}
```
